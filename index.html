--!strict
type Pool = { [string]: number }

local rng = Random.new()

local function roll(pool: Pool): string
  local total = 0
  for _, weight in pool do
    total += weight
  end

  local pick = rng:NextNumber() * total
  local last = ""
  for id, weight in pool do
    last = id
    pick -= weight
    if pick < 0 then return id end
  end
  return last
end
